cmake_minimum_required(VERSION 3.20)
project(PortabilityTest C)

# -----------------------------
# Build matrix inputs (set by run_all.py, defaults for plain builds)
# -----------------------------
set(C_STD "c11" CACHE STRING "C standard: c99 or c11")
set(OPT_LEVEL "-O0" CACHE STRING "Optimization level: -O0 or -O2")
set(COMPILER_NAME "" CACHE STRING "Expected compiler: gcc, clang or msvc (optional)")
set(SANITIZERS "" CACHE STRING "Sanitizers to enable, e.g. address,undefined or thread (MSVC: address only)")

# Fail if the compiler is not the one the run is labelled with
set(_expected_ids_gcc "GNU")
set(_expected_ids_clang "Clang|AppleClang")
set(_expected_ids_msvc "MSVC")
if (COMPILER_NAME AND NOT CMAKE_C_COMPILER_ID MATCHES "^(${_expected_ids_${COMPILER_NAME}})$")
    message(FATAL_ERROR "COMPILER_NAME=${COMPILER_NAME} but the compiler is ${CMAKE_C_COMPILER_ID}")
endif()

# C standard: strict ISO mode (-std=c99, not -std=gnu99)
if (C_STD STREQUAL "c99")
    set(_std_version 199901L)
    if (MSVC)
        message(FATAL_ERROR "MSVC has no C99 mode; use C_STD=c11")
    endif()
elseif (C_STD STREQUAL "c11")
    set(_std_version 201112L)
else()
    message(FATAL_ERROR "Unsupported C_STD '${C_STD}' (expected c99 or c11)")
endif()

string(REGEX REPLACE "^c" "" CMAKE_C_STANDARD "${C_STD}")
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)

# Optimization: OPT_LEVEL is the only source, so drop the per-config defaults
# (e.g. Release's /O2 /DNDEBUG) and keep asserts on in every build
foreach (_config DEBUG RELEASE RELWITHDEBINFO MINSIZEREL)
    set(CMAKE_C_FLAGS_${_config} "")
    set(CMAKE_CXX_FLAGS_${_config} "")
endforeach()

if (OPT_LEVEL STREQUAL "-O0")
    set(_optimized 0)
    set(_msvc_opt /Od)
elseif (OPT_LEVEL STREQUAL "-O2")
    set(_optimized 1)
    set(_msvc_opt /O2)
else()
    message(FATAL_ERROR "Unsupported OPT_LEVEL '${OPT_LEVEL}' (expected -O0 or -O2)")
endif()

add_executable(hammer_ironclib 
    main.c
    tests/inline_func_wrapper_a.c
    tests/inline_func_wrapper_b.c
    tests/opaque_structs.c
    tests/co_job_test_funcs.c
    # Each includes one library header first, then uses its macros, so the build fails
    # if a header (or a macro it defines) depends on something it does not include itself
    tests/standalone_typenum.c
    tests/standalone_opaque_storage.c
    tests/standalone_result.c
    tests/standalone_num_cast.c
    tests/standalone_co_job.c
    tests/standalone_memory_and_loops.c
)

# Every library header must also compile as the very first and only include of a translation unit.
# One tiny source per header is generated in the build directory, so new headers are covered automatically.
file(GLOB _ihc_library_headers CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/../ironclib/*.h)
foreach (_header ${_ihc_library_headers})
    get_filename_component(_header_name ${_header} NAME_WE)
    set(_standalone_source ${CMAKE_CURRENT_BINARY_DIR}/standalone/${_header_name}_alone.c)
    # file(CONFIGURE) only rewrites the file when its content changes, so builds stay incremental
    file(CONFIGURE OUTPUT ${_standalone_source} CONTENT
"// Generated by CMakeLists.txt: ${_header_name}.h must compile as the very first and only include
#include \"ironclib/${_header_name}.h\"
typedef int ihc_${_header_name}_compiles_alone;
")
    target_sources(hammer_ironclib PRIVATE ${_standalone_source})
endforeach()

target_include_directories(hammer_ironclib PRIVATE
    ${CMAKE_CURRENT_SOURCE_DIR}
    ${CMAKE_CURRENT_SOURCE_DIR}/tests
    ${CMAKE_CURRENT_SOURCE_DIR}/..
    ${CMAKE_CURRENT_SOURCE_DIR}/../ironclib
)

# main.c checks these against __STDC_VERSION__ / __OPTIMIZE__ and fails the build on mismatch
target_compile_definitions(hammer_ironclib PRIVATE
    IHC_EXPECTED_STDC_VERSION=${_std_version}
    IHC_EXPECTED_OPTIMIZED=${_optimized}
)

# Strict ISO mode makes glibc hide POSIX declarations (e.g. nanosleep) used by the pthread backend.
# Apple's libc declares them by default, and defining _POSIX_C_SOURCE there would hide
# Darwin-specific declarations instead, so macOS is left alone.
if (NOT WIN32 AND NOT APPLE)
    target_compile_definitions(hammer_ironclib PRIVATE _POSIX_C_SOURCE=200809L)
endif()

# libm for <math.h> functions (nextafter, ldexp) used by the num cast tests
if (UNIX AND NOT APPLE)
    target_link_libraries(hammer_ironclib PRIVATE m)
endif()

if (CMAKE_C_COMPILER_ID MATCHES "Clang|GNU")
    target_compile_options(hammer_ironclib PRIVATE
        ${OPT_LEVEL}
        -Wall
        -Wextra
        -Wpedantic
        -Werror

        # macro-generated static_assert typedef noise ----
        -Wno-unused-local-typedefs
    )
endif()

if (MSVC)
    target_compile_options(hammer_ironclib PRIVATE
        ${_msvc_opt}
        /W4
        /WX
        /wd4127  # macro pattern (do { } while(0))
    )
endif()

# -----------------------------
# C++ compatibility (compile-only)
# -----------------------------
# A C++11 source includes every library and premade header and uses their main macros, which checks
# the extern "C" guards and C++-valid code, plus the C++-only branches (IC_STATIC_ASSERT, IC_ALIGNAS,
# IC_ALIGNOF, decltype in IC_TRY_RETURN_ERR_AS) that no C build selects. It is an object library that is
# never linked, so the test executable stays pure C. Skipped when no C++ compiler is available.
include(CheckLanguage)
check_language(CXX)
if (CMAKE_CXX_COMPILER)
    enable_language(CXX)

    add_library(ihc_cpp_compat OBJECT tests/cpp_compat.cpp)
    target_include_directories(ihc_cpp_compat PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/..)
    if (NOT WIN32 AND NOT APPLE)
        target_compile_definitions(ihc_cpp_compat PRIVATE _POSIX_C_SOURCE=200809L) # same reason as the C target
    endif()
    set_target_properties(ihc_cpp_compat PROPERTIES
        CXX_STANDARD 11
        CXX_STANDARD_REQUIRED ON
        CXX_EXTENSIONS OFF
    )

    if (CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
        target_compile_options(ihc_cpp_compat PRIVATE ${OPT_LEVEL} -Wall -Wextra -Wpedantic -Werror)
    elseif (MSVC)
        # Deliberately WITHOUT /Zc:preprocessor: MSVC C++ uses its traditional preprocessor by default,
        # so this checks that every macro also works there (the C target gets the conforming one via /std:c11)
        target_compile_options(ihc_cpp_compat PRIVATE ${_msvc_opt} /W4 /WX /wd4127) # same as the C target
    endif()

    # Building the test executable always builds (and so checks) the C++ source too
    add_dependencies(hammer_ironclib ihc_cpp_compat)
else()
    message(STATUS "No C++ compiler found, skipping the C++ compatibility check")
endif()

# -----------------------------
# Compile-fail tests (configure time)
# -----------------------------
# Some guarantees are compile errors (late configuration tags, false static asserts), which a
# normal test cannot observe. Each source in tests/compile_fail/ is compiled twice:
#   1. with IHC_COMPILE_FAIL_CONTROL defined, which must compile (so the setup itself is valid), and
#   2. without it, which must fail, and when given, with the expected message in the compiler output.
# A broken guarantee stops the configure step with an explanation.
option(IHC_COMPILE_FAIL_TESTS "Run the compile-fail tests at configure time" ON)

if (IHC_COMPILE_FAIL_TESTS)
    set(CMAKE_TRY_COMPILE_TARGET_TYPE STATIC_LIBRARY) # compile only, nothing to link

    set(_ihc_compile_fail_definitions "")
    if (NOT WIN32 AND NOT APPLE)
        list(APPEND _ihc_compile_fail_definitions -D_POSIX_C_SOURCE=200809L) # same reason as above
    endif()

    function(ihc_expect_compile_error name expected_message)
        set(_source ${CMAKE_CURRENT_SOURCE_DIR}/tests/compile_fail/${name}.c)
        set(_bin ${CMAKE_CURRENT_BINARY_DIR}/compile_fail/${name})

        try_compile(${name}_control_compiled ${_bin}/control
            SOURCES ${_source}
            CMAKE_FLAGS "-DINCLUDE_DIRECTORIES=${CMAKE_CURRENT_SOURCE_DIR}/.."
            COMPILE_DEFINITIONS ${_ihc_compile_fail_definitions} -DIHC_COMPILE_FAIL_CONTROL
            C_STANDARD ${CMAKE_C_STANDARD}
            C_STANDARD_REQUIRED ON
            C_EXTENSIONS OFF
            OUTPUT_VARIABLE _control_output
        )
        if (NOT ${name}_control_compiled)
            message(FATAL_ERROR "Compile-fail test '${name}': the control version must compile, but it failed:\n${_control_output}")
        endif()

        try_compile(${name}_compiled ${_bin}/test
            SOURCES ${_source}
            CMAKE_FLAGS "-DINCLUDE_DIRECTORIES=${CMAKE_CURRENT_SOURCE_DIR}/.."
            COMPILE_DEFINITIONS ${_ihc_compile_fail_definitions}
            C_STANDARD ${CMAKE_C_STANDARD}
            C_STANDARD_REQUIRED ON
            C_EXTENSIONS OFF
            OUTPUT_VARIABLE _output
        )
        if (${name}_compiled)
            message(FATAL_ERROR "Compile-fail test '${name}': must fail to compile, but it compiled")
        endif()

        if (NOT expected_message STREQUAL "")
            string(FIND "${_output}" "${expected_message}" _message_position)
            if (_message_position EQUAL -1)
                message(FATAL_ERROR "Compile-fail test '${name}': failed to compile, but without the expected error '${expected_message}':\n${_output}")
            endif()
        endif()

        message(STATUS "Compile-fail test '${name}': passed")
    endfunction()

    ihc_expect_compile_error(late_c11_threads_define "IC_USE_C11_THREADS_AND_ATOMICS was defined after")
    ihc_expect_compile_error(parallel_work_after_concurrency "parallel_work.h must be included before")
    # Only the native static assert (C11+) prints the message; C99 builds use the portable fallback
    if (C_STD STREQUAL "c99")
        set(_static_assert_message "")
    else()
        set(_static_assert_message "ihc expected static assert failure")
    endif()
    ihc_expect_compile_error(static_assert_false "${_static_assert_message}")
    ihc_expect_compile_error(portable_static_assert_false "") # negative array size, wording differs per compiler
endif()

# Sanitizers (set by run_all.py from the SANITIZERS environment variable)
if (SANITIZERS)
    if (CMAKE_C_COMPILER_ID MATCHES "Clang|GNU")
        # -fno-sanitize-recover makes UBSan findings fail the run instead of only printing
        target_compile_options(hammer_ironclib PRIVATE
            -fsanitize=${SANITIZERS}
            -fno-sanitize-recover=all
            -fno-omit-frame-pointer
        )
        target_link_options(hammer_ironclib PRIVATE -fsanitize=${SANITIZERS})
    elseif (MSVC)
        if (NOT SANITIZERS STREQUAL "address")
            message(FATAL_ERROR "MSVC only supports SANITIZERS=address, got '${SANITIZERS}'")
        endif()
        target_compile_options(hammer_ironclib PRIVATE /fsanitize=address)
    else()
        message(FATAL_ERROR "SANITIZERS is not supported for compiler ${CMAKE_C_COMPILER_ID}")
    endif()
endif()
